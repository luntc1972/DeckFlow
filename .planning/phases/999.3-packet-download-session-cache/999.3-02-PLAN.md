The controller passes through only the typed Request DTO; ALL normalization + extraction logic lives ON THE SERVICE.

This makes write↔read parity STRUCTURAL — both code paths construct the bag from the same private data.

    /// <summary>
    /// Composes the D-01 cache-input field bag and returns the canonical PacketSessionCache key
    /// for this request. The implementation re-runs the same private LoadDeckEntriesAsync path
    /// BuildAsync uses and derives the commander from parsed entries via the same heuristic
    /// (FirstOrDefault commander board → Moxfield-ordering fallback). Returns null on load failure
    /// or empty deck — in that case the controller's download handler falls through to BuildAsync
    /// silently (D-11). Both BuildAsync (write side) and DeckController download handlers (read side)
    /// derive the cache key by composing the field bag in-place with the same data this helper uses,
    /// so write↔read parity is enforced by code locality.
    /// </summary>
    /// <param name="request">Deck-analysis request to derive a cache key from.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>64-character lowercase SHA-256 hex string, or null if the key cannot be computed.</returns>
    public async Task<string?> TryComputeCacheKeyAsync(DeckAnalysisRequest request, CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (string.IsNullOrWhiteSpace(request.DeckSource))
        {
            return null;
        }

        List<DeckEntry> entries;
        try
        {
            entries = await LoadDeckEntriesAsync(request.DeckSource, cancellationToken).ConfigureAwait(false);
        }
        catch (Exception ex) when (ex is InvalidOperationException or DeckParseException or HttpRequestException)
        {
            return null;
        }

        if (entries.Count == 0)
        {
            return null;
        }

        // Mirror BuildAsync's commander derivation (DeckAnalysisPacketService.cs:226-271) BUT
        // SKIP the Scryfall ValidateCommanderAsync round-trip (the very call the cache exists
        // to eliminate). The unvalidated raw name is sufficient for cache-key parity: BuildAsync
        // ultimately composes the result using the validated name, but the write side computes
        // its cache key using the SAME unvalidated raw name path, so both sides agree.
        var deckEntries = entries
            .Where(entry =>
                !string.Equals(entry.Board, "maybeboard", StringComparison.OrdinalIgnoreCase)
                && !string.Equals(entry.Board, "sideboard", StringComparison.OrdinalIgnoreCase))
            .ToList();

        if (deckEntries.Count == 0)
        {
            return null;
        }

        var commanderName = deckEntries
            .FirstOrDefault(entry => string.Equals(entry.Board, "commander", StringComparison.OrdinalIgnoreCase))
            ?.Name;

        if (commanderName is null && entries.Count > 0)
        {
            var leadingOneOfs = entries
                .TakeWhile(entry => entry.Quantity == 1
                    && !string.Equals(entry.Board, "maybeboard", StringComparison.OrdinalIgnoreCase)
                    && !string.Equals(entry.Board, "sideboard", StringComparison.OrdinalIgnoreCase))
                .Take(1)
                .ToList();
            if (leadingOneOfs.Count > 0)
            {
                commanderName = leadingOneOfs[0].Name;
            }
        }

        var normalizedDeckSource = BuildCanonicalDeckSourceText(entries);

        var cacheInputs = new DeckAnalysisCacheInputs(
            Commander: commanderName ?? string.Empty,
            NormalizedDeckSource: normalizedDeckSource,
            IncludeCardVersions: request.IncludeCardVersions,
            IncludeSideboardInAnalysis: request.IncludeSideboardInAnalysis,
            IncludeMaybeboardInAnalysis: request.IncludeMaybeboardInAnalysis,
            TargetAiPlatformKey: request.TargetAiPlatform,
            SelectedQuestionIds: (request.SelectedAnalysisQuestions ?? new List<string>())
                .OrderBy(static id => id, StringComparer.Ordinal)
                .ToArray());

        return PacketSessionCache.ComputeKey(cacheInputs);
    }

    private static string BuildCanonicalDeckSourceText(IReadOnlyList<DeckEntry> entries)
    {
        // Emit a stable text representation of the loaded deck so URL-mode and paste-mode
        // collapse to one cache key (D-02). Sort by board then name then set then collector
        // number for byte-stable output across runs.
        var builder = new StringBuilder();
        foreach (var entry in entries
            .OrderBy(e => e.Board ?? string.Empty, StringComparer.OrdinalIgnoreCase)
            .ThenBy(e => e.Name ?? string.Empty, StringComparer.OrdinalIgnoreCase)
            .ThenBy(e => e.SetCode ?? string.Empty, StringComparer.OrdinalIgnoreCase)
            .ThenBy(e => e.CollectorNumber ?? string.Empty, StringComparer.OrdinalIgnoreCase))
        {
            builder.Append(entry.Board ?? string.Empty).Append('|')
                   .Append(entry.Quantity).Append('|')
                   .Append(entry.Name ?? string.Empty).Append('|')
                   .Append(entry.SetCode ?? string.Empty).Append('|')
                   .Append(entry.CollectorNumber ?? string.Empty).Append('\n');
        }
        return builder.ToString();
    }
