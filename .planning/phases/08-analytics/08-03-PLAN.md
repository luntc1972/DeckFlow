    using DeckFlow.Web.Services.Analytics;
    using Microsoft.AspNetCore.Hosting;

    namespace DeckFlow.Web.Extensions;

    /// <summary>
    /// DI registration for Phase 8 analytics. Mirrors HarvestServiceCollectionExtensions
    /// (singleton-and-hosted-service for the BackgroundService; lazy IServiceProvider for the
    /// store per the Phase 7.1 dc66a38 errata pattern — D-14).

    public static class AnalyticsServiceCollectionExtensions
    {
        public static IServiceCollection AddDeckFlowAnalytics(this IServiceCollection services, IWebHostEnvironment env)
        {
            ArgumentNullException.ThrowIfNull(services);
            ArgumentNullException.ThrowIfNull(env);

            // Buffer: singleton, no dependencies (D-08).
            services.AddSingleton<RequestMetricsBuffer>();

            // Store: singleton with lazy IServiceProvider per D-14 (Phase 7.1 errata pattern).
            services.AddSingleton<IRequestMetricsStore>(sp => new RequestMetricsStore(

                sp));

            // Flusher: singleton + hosted service (mirrors HarvestScheduleCache pattern).

            // Salt accessor: singleton string holder, populated at startup before middleware runs.
            services.AddSingleton<AnalyticsSaltAccessor>();

            // Middleware: scoped per request (IMiddleware contract).
            services.AddScoped<DeckFlow.Web.Infrastructure.AnalyticsMiddleware>();

            return services;
        }
    }

    /// <summary>
    /// Holds the resolved IP-hash salt. Populated once at startup so the analytics middleware
    /// does not pay async DB I/O on the hot path. Volatile read/write avoids torn reads under
    /// the highly unlikely racing-startup-vs-first-request scenario.
    /// </summary>
    public sealed class AnalyticsSaltAccessor
    {
        private string? _salt;
        public string? Salt => System.Threading.Volatile.Read(ref _salt);
        public void SetSalt(string salt)
        {
            ArgumentNullException.ThrowIfNull(salt);
            System.Threading.Volatile.Write(ref _salt, salt);
        }
    }
