  comparator TEXT NOT NULL, condition TEXT NULL, clip_ts INT NULL,
  source_clip TEXT NOT NULL, confidence DOUBLE PRECISION NOT NULL,
  card_reference TEXT NULL, card_grounded BOOLEAN NULL, sort_order INT NOT NULL DEFAULT 0
