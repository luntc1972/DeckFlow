**Core Process.Start pattern (stdin + stdout)** — extend `FfmpegAudioChunker.cs` lines 60-82 with stdin/stdout:
```csharp
    // IMPORTANT: Start ReadToEndAsync on BOTH stdout and stderr BEFORE WaitForExitAsync
    // to prevent OS pipe buffer deadlock (see Pitfall 2 in RESEARCH.md).
    // IMPORTANT: Close stdin BEFORE reading stdout so the CLI sees EOF and returns
    // (see Pitfall 1 in RESEARCH.md).
    private static async Task<string> RunProcessAsync(
        string fileName,
        string arguments,
        string prompt,
        CancellationToken ct)
    {
        var startInfo = new ProcessStartInfo(fileName)
        {
            Arguments = arguments,   // from ResolveCommand — the CLI flags (e.g. claude: -p --output-format json --allowedTools "")
            RedirectStandardInput = true,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true,
        };

        using var process = Process.Start(startInfo)
            ?? throw new InvalidOperationException($"'{fileName}' process failed to start.");

        await process.StandardInput.WriteAsync(prompt, ct).ConfigureAwait(false);
        process.StandardInput.Close();   // signal EOF — without this the CLI hangs forever

        var stdoutTask = process.StandardOutput.ReadToEndAsync(ct);
        var stderrTask = process.StandardError.ReadToEndAsync(ct);
        await process.WaitForExitAsync(ct).ConfigureAwait(false);
        var stdout = await stdoutTask.ConfigureAwait(false);
        var stderr = await stderrTask.ConfigureAwait(false);

        if (process.ExitCode != 0)
        {
            throw new InvalidOperationException(
                $"'{fileName}' exited {process.ExitCode}. stderr: {Tail(stderr)}");
        }

        return stdout;
    }

    // Mirrors FfmpegAudioChunker.Tail (lines 87-95) exactly:
    private static string Tail(string text)
    {
        if (string.IsNullOrWhiteSpace(text) || text.Length <= ErrorTailLength)
        {
            return text;
        }

        return text[^ErrorTailLength..];
    }
```

**Platform + command resolution pattern** — new for this service (no existing analog; per RESEARCH.md Critical Unknown 3):
```csharp
    private static (string fileName, string arguments) ResolveCommand(string provider)
    {
        var envOverride = Environment.GetEnvironmentVariable(CliCommandEnvironmentKey);
        if (!string.IsNullOrWhiteSpace(envOverride))
        {
            var parts = envOverride.Trim().Split(' ', 2);
            return (parts[0], parts.Length > 1 ? parts[1] : string.Empty);
        }

        if (OperatingSystem.IsLinux() || OperatingSystem.IsMacOS())
        {
            return provider switch
            {
                "claude" => ("claude", "-p --output-format json --allowedTools \"\""),
                "codex"  => ("codex", "exec - --sandbox read-only"),
                _ => throw new NotSupportedException($"Unknown provider '{provider}'."),
            };
        }

        // Why: on Windows, claude/codex are not in the Windows PATH on this machine;
        // .cmd shims also cannot be spawned with UseShellExecute=false without cmd.exe /c.
        // Require operator to set DECKFLOW_LLM_CLI_COMMAND explicitly on Windows.
        throw new InvalidOperationException(
            $"Cannot auto-detect '{provider}' CLI on Windows. " +
            $"Set {CliCommandEnvironmentKey} to the full command, e.g.: " +
            $"'wsl.exe {provider} ...' or 'cmd.exe /c {provider}.cmd ...'");
    }
```

**Envelope unwrap + fence-strip + brace-match pattern** — new for this service; based on RESEARCH.md verified output contracts:
```csharp
    private string ExtractModelText(string stdout)
    {
        if (string.Equals(_provider, "claude", StringComparison.OrdinalIgnoreCase))
        {
            using var doc = JsonDocument.Parse(stdout);
            var root = doc.RootElement;
            var isError = root.TryGetProperty("is_error", out var errProp) && errProp.GetBoolean();
            if (isError)
            {
                var errMsg = root.TryGetProperty("result", out var r) ? r.GetString() : "unknown";
                throw new InvalidOperationException($"claude returned is_error=true: {errMsg}");
            }
            return root.GetProperty("result").GetString()
                ?? throw new InvalidOperationException("claude .result field is null.");
        }

        // codex: stdout IS the model text (no envelope)
        return stdout.Trim();
    }

    private static string FenceStrip(string text)
    {
        // Strip ```json ... ``` or ``` ... ``` fences
        var stripped = text.Trim();
        if (stripped.StartsWith("```", StringComparison.Ordinal))
        {
            var firstNewline = stripped.IndexOf('\n');
            if (firstNewline >= 0)
            {
                stripped = stripped[(firstNewline + 1)..];
            }
        }
        if (stripped.EndsWith("```", StringComparison.Ordinal))
        {
            stripped = stripped[..^3];
        }
        return stripped.Trim();
    }

    private static string BraceMatch(string text)
    {
        var start = text.IndexOf('{');
        var end = text.LastIndexOf('}');
        if (start < 0 || end < start)
        {
            throw new InvalidOperationException("No JSON object found in model output.");
        }
        return text[start..(end + 1)];
    }
```

**Retry-loop pattern per extraction** — new for this service; retry shape mirrors existing `DistillVideoAsync` outer try/catch:
```csharp
    private async Task<T> ExtractWithRetryAsync<T>(
        string prompt,
        Action<T> validate,
        CancellationToken ct)
    {
        Exception? last = null;
        for (var attempt = 0; attempt < MaxRetries; attempt++)
        {
            try
            {
                var (fileName, arguments) = ResolveCommand(_provider);
                var stdout = await _runProcess(fileName, arguments, prompt, ct).ConfigureAwait(false);
                var modelText = ExtractModelText(stdout);
                var json = BraceMatch(FenceStrip(modelText));
                var payload = JsonSerializer.Deserialize<T>(json, JsonOpts)
                    ?? throw new InvalidOperationException("Deserialized to null.");
                validate(payload);
                return payload;
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested)
            {
                throw;
            }
            catch (Exception ex)
            {
                last = ex;
            }
        }
        throw new InvalidOperationException(
            $"CLI extraction failed after {MaxRetries} attempts.", last);
    }
```

namespace DeckFlow.Core.Integration;

/// <summary>
/// Resolves the configured LLM distillation provider.
/// </summary>
public static class LlmDistillationProviderFactory
{
    /// <summary>
    /// Environment variable used to select the LLM distillation provider.
    /// </summary>
    public const string EnvironmentVariableName = "DECKFLOW_LLM_PROVIDER";

    private const string OpenAiProvider = "openai";
    private const string ClaudeProvider = "claude";
    private const string CodexProvider = "codex";

    /// <summary>
    /// Resolves an LLM distillation service from the environment toggle.
    /// </summary>
    /// <param name="httpClient">HTTP client for the OpenAI provider.</param>
    /// <returns>The configured LLM distillation service.</returns>
    public static ILlmDistillationService Resolve(HttpClient httpClient)
        => Resolve(Environment.GetEnvironmentVariable(EnvironmentVariableName), httpClient);

    /// <summary>
    /// Resolves an LLM distillation service from a provider value.
    /// </summary>
    /// <param name="providerEnvValue">Provider value read from <see cref="EnvironmentVariableName"/>.</param>
    /// <param name="httpClient">HTTP client for the OpenAI provider.</param>
    /// <returns>The configured LLM distillation service.</returns>
    /// <exception cref="NotSupportedException">Thrown when the provider value is unsupported.</exception>
    public static ILlmDistillationService Resolve(string? providerEnvValue, HttpClient httpClient)
    {
        ArgumentNullException.ThrowIfNull(httpClient);
        var provider = string.IsNullOrWhiteSpace(providerEnvValue)
            ? OpenAiProvider
            : providerEnvValue.Trim();

        if (string.Equals(provider, OpenAiProvider, StringComparison.OrdinalIgnoreCase))
        {
            return new LlmDistillationService(httpClient);
        }

        if (string.Equals(provider, ClaudeProvider, StringComparison.OrdinalIgnoreCase))
        {
            return new CliLlmDistillationService(ClaudeProvider);
        }

        if (string.Equals(provider, CodexProvider, StringComparison.OrdinalIgnoreCase))
        {
            return new CliLlmDistillationService(CodexProvider);
        }

        throw new NotSupportedException(
            $"Unsupported {EnvironmentVariableName} '{provider}'. Supported: openai, claude, codex.");
    }
}
```

Key conventions to copy from `TranscriptProviderFactory.cs`:
- `public const string EnvironmentVariableName` (not readonly, not private — tests reference it by name)
- `ArgumentNullException.ThrowIfNull(httpClient)` at top of the overload that takes the value
- `string.IsNullOrWhiteSpace` → default branch (not null-check alone)
- `string.Equals(..., StringComparison.OrdinalIgnoreCase)` for each provider string comparison
- `NotSupportedException` message lists all supported values
