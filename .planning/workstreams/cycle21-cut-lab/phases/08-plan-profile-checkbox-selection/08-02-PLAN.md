DeckFlow.Web/Services/CutLab/CutLabPlanAffinityResolver.cs, namespace DeckFlow.Web.Services.CutLab:
  public sealed record CutLabPlanAffinity
    IReadOnlyList<string> OnPlanThemes      // display names of CHECKED themes this card appears in
    IReadOnlyList<string> OffPlanThemes     // display names of UNCHECKED themes this card appears in
    IReadOnlyList<string> OnPlanStrategies  // display names of CHECKED generic strategies this card proxies
    int Score                               // min(OnPlanThemes.Count + OnPlanStrategies.Count, OnPlanScoreCap)
    bool IsOnPlan => Score > 0
    static CutLabPlanAffinity Neutral { get; }   // all empty, Score 0
  public static class CutLabPlanAffinityResolver
    internal const int OnPlanScoreCap = 3
    static IReadOnlyDictionary<string, CutLabPlanAffinity> ResolveAll(
        IReadOnlyList<CutLabAnalyzedCard> pool,
        CutLabPlanProfile? planProfile,
        IReadOnlyDictionary<string, IReadOnlyList<string>> themeCardNamesBySlug,
        IReadOnlyList<CutLabCommanderTheme> allKnownThemes)
      // returns a dictionary keyed by CutLabCardNames.Normalize(card.Name) using CutLabCardNames.Comparer
    static CutLabPlanAffinity For(IReadOnlyDictionary<string, CutLabPlanAffinity> affinities, string cardName)
      // normalizing lookup returning Neutral on miss
</interfaces>

  <behavior>
    - Theme membership: a pool card whose normalized name equals a normalized entry in
      `themeCardNamesBySlug[checkedTheme.Slug]` gains that theme's `DisplayName` in `OnPlanThemes` and
      +1 raw score.
    - Off-plan themes: the same match against a theme present in `allKnownThemes` but absent from
      `planProfile.CommanderThemes` puts that theme's `DisplayName` in `OffPlanThemes` and contributes
      NOTHING to `Score`.
    - Strategy membership: for each checked slug resolvable through `DeckPlanStrategyCatalog.TryGetBySlug`,
      if `MatchesCategories` is true for the card's `Categories`, the entry's `DisplayName` joins
      `OnPlanStrategies` and adds +1 raw score.
    - Union: a card matched only by a theme, and a card matched only by a strategy, both come back with
      `IsOnPlan` true. Neither layer gates the other.
    - Cap: `Score` is `Math.Min(rawScore, OnPlanScoreCap)`. A card matching two themes and two strategies
      scores `OnPlanScoreCap`, not 4.
    - No-op: a null `planProfile`, or one with empty `GenericStrategies` AND empty `CommanderThemes`,
      resolves every pool card to `CutLabPlanAffinity.Neutral`.
    - Unknown slug: a checked strategy slug not in the catalog is ignored silently and contributes no score.
    - Unknown theme slug: a checked theme with no entry in `themeCardNamesBySlug` (its card list was
      never fetched) contributes no membership and does not throw.
    - Determinism: `OnPlanThemes`, `OffPlanThemes` and `OnPlanStrategies` are ordered — themes by their
      order in the supplied theme collections, strategies by `DeckPlanStrategyCatalog.Entries` order —
      so repeated runs over identical input produce identical lists.
  </behavior>
